interface TreeDir {
  name: string
  path: string // full dir path, e.g. "src/components"
  dirs: Map<string, TreeDir>
  files: FileNode[]

function buildTree(nodes: FileNode[]): TreeDir {
  const root: TreeDir = { name: '', path: '', dirs: new Map(), files: [] }
  for (const node of nodes) {
    const parts = node.id.split('/')
    parts.pop() // drop filename; keep dir segments
    let dir = root
    for (const part of parts) {
      if (!dir.dirs.has(part)) {
        const path = dir.path ? `${dir.path}/${part}` : part
        dir.dirs.set(part, { name: part, path, dirs: new Map(), files: [] })
      }
      dir = dir.dirs.get(part)!
    }
    dir.files.push(node)
  }
  return root
}

  return out
}

/** Trail difficulty from its stops' interestingness — blend of average + hardest. */
function difficulty(stops: Stop[]): Difficulty | null {
  if (stops.length === 0) return null
  const scores = stops.map((s) => s.node.score)
  const avg = scores.reduce((a, b) => a + b, 0) / scores.length
  const max = Math.max(...scores)
  const rating = 0.6 * avg + 0.4 * max
  if (rating < 0.35) return { icon: '●', label: 'Easy', cls: 'diff-easy', rating }
