190	/** BFS from entrypoints along out-edges (dynamic included). Unreached = null. */
191	function bfsDepth(files: string[], edges: Edge[], entrypoints: string[]): Map<string, number | null> {
192	  const adj = new Map<string, string[]>()
193	  for (const e of edges) {
194	    if (!adj.has(e.from)) adj.set(e.from, [])
195	    adj.get(e.from)!.push(e.to)
196	  }
197	  const depth = new Map<string, number | null>()
198	  for (const f of files) depth.set(f, null)
199	  const queue: string[] = []
200	  for (const ep of entrypoints) { depth.set(ep, 0); queue.push(ep) }
201	  while (queue.length) {
202	    const cur = queue.shift()!
203	    const d = depth.get(cur)!
204	    for (const next of adj.get(cur) ?? []) {
205	      if (depth.get(next) === null) { depth.set(next, (d ?? 0) + 1); queue.push(next) }
206	    }
207	  }
208	  return depth
209	}

211	/**
212	 * Cheap cycle indicator: count of DFS back edges. Iterative (an explicit stack)
213	 * so a deep first-party graph can't overflow the call stack and crash analyze.
214	 */
215	function countCycles(files: string[], edges: Edge[]): number {
216	  const adj = new Map<string, string[]>()
217	  for (const e of edges) {
218	    if (!adj.has(e.from)) adj.set(e.from, [])
219	    adj.get(e.from)!.push(e.to)
220	  }
221	  const state = new Map<string, 0 | 1 | 2>() // 0 unvisited, 1 in-stack, 2 done
222	  let back = 0
223	  for (const start of files) {
224	    if ((state.get(start) ?? 0) !== 0) continue
225	    const stack: { node: string; i: number }[] = [{ node: start, i: 0 }]
226	    state.set(start, 1)
227	    while (stack.length) {
228	      const frame = stack[stack.length - 1]
229	      const neighbors = adj.get(frame.node) ?? []
230	      if (frame.i < neighbors.length) {
231	        const m = neighbors[frame.i++]
232	        const s = state.get(m) ?? 0
233	        if (s === 1) back++
234	        else if (s === 0) { state.set(m, 1); stack.push({ node: m, i: 0 }) }
235	      } else {
236	        state.set(frame.node, 2)
237	        stack.pop()
238	      }
239	    }
240	  }
241	  return back
242	}
